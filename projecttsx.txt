import React from 'react'
import { Link } from 'react-router-dom'

// Json Imports
import gdjson from '../json/gd.json'
import sdejson from '../json/sde.json'

interface ProjectItem {
  title: string
  genre: string
  thumbnail: string
}

const formatTitleForUrl = (title: string) =>
  title.trim().toLowerCase().replace(/\s+/g, '-')

const ProjectSection: React.FC<{
  eyebrow: string
  heading: string
  items: ProjectItem[]
}> = ({ eyebrow, heading, items }) => (
  <section className="mb-24 last:mb-0">
    <div className="mb-10">
      <p className="ray-olsen text-xs tracking-[0.2em] uppercase text-[#8C897F] mb-2">
        {eyebrow}
      </p>
      <h2 className="font-serif text-4xl md:text-5xl text-[#1B1B18]">
        {heading}
      </h2>
    </div>

    <div className="border-t border-[#E4E1D8]">
      {items.map((item, i) => (
        <Link
          key={item.title}
          to={`/projects/${formatTitleForUrl(item.title)}`}
          className="group flex items-center gap-6 md:gap-10 py-6 border-b border-[#E4E1D8] transition-colors duration-300 hover:bg-[#F2F0EA]/60"
        >
          <span className="font-mono text-xs text-[#5B6B4F] w-6 shrink-0">
            {String(i + 1).padStart(2, '0')}
          </span>

          <div className="w-16 h-16 md:w-20 md:h-20 shrink-0 overflow-hidden ">
            <img
              src={item.thumbnail}
              alt={item.title}
              className="w-full h-full object-cover grayscale group-hover:grayscale-0 scale-100 group-hover:scale-105 transition-all duration-500 ease-out"
            />
          </div>

          <div className="flex-1 min-w-0">
            <h3 className="font-serif text-xl md:text-2xl text-[#1B1B18] truncate">
              {item.title}
            </h3>
            <p className="text-sm text-[#8C897F]">{item.genre}</p>
          </div>

          <span className="font-mono text-sm text-[#1B1B18] shrink-0 opacity-0 -translate-x-2 group-hover:opacity-100 group-hover:translate-x-0 transition-all duration-300">
            View →
          </span>
        </Link>
      ))}
    </div>
  </section>
)

const Project: React.FC = () => {
  return (
    <div className="bg-[#FAF9F6] min-h-screen">
      <div className="max-w-4xl mx-auto px-6 md:px-8 py-16 md:py-24">
        <ProjectSection
          eyebrow="Selected work"
          heading="Web Projects"
          items={sdejson}
        />
        <ProjectSection
          eyebrow="Off the clock"
          heading="Games I've Made"
          items={gdjson}
        />
      </div>
    </div>
  )
}

export default Project
